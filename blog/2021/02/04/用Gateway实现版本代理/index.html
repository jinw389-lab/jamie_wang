用Gateway实现版本代理

Posted by jamie Blog on 2021-02-04 · 阅读 -- 次

1. 需求

有这么一个需求，由于项目存在不同的版本，我们能人为的控制用户使用不同的版本；
现在要实现一个转发服务，当请求来时，获取不要参数，通过不同参数转发到不同环境。画一张图可能更加容易理解。

请求发送，参数中sn=0001,转发服务得到sn的值，通过业务逻辑得到版本号，若是V2，则获取V2的目标服务地址，饭后转发过去。

需求还是很简单的，但是存在几个问题： 问题1：跨环境调用
这样的话，就不能通过注册中心来的applicaiton name来实现了。所以我们可以考虑通过ip地址进行转发。这样部署集群也同样可以，转发服务也不用要求必须部署在哪个环境。

问题2：正因为问题1，所以配置路由也存在问题，转发地址的不明确性，这个问题只能设置Filter,动态去改变目标路由地址。

    private static final String GATEWAY_ROUTE_ATTR = "org.springframework.cloud.gateway.support.ServerWebExchangeUtils.gatewayRoute";
    private static final String GATEWAY_ROUTE_URI = "org.springframework.cloud.gateway.support.ServerWebExchangeUtils.gatewayRequestUrl";
    private static final String HEADER_BODY = "bodyStr";
    private static final String HEADER_VERSION = "version";

 @Bean
    @Order(-1)
    public GlobalFilter preFilter() {
        return (exchange, chain) -> {
            ServerHttpRequest req = exchange.getRequest();

            String version = req.getHeaders().getFirst(HEADER_VERSION);
            String bodyStr = req.getHeaders().getFirst(HEADER_BODY);
            List<String> toUriList = deviceProxyGatewayProperties.getRoutes().get(version);
            Random random = new Random();
            String toUri = toUriList.get(random.nextInt(toUriList.size())); //"http://localhost:19324";
            String rawPath = req.getURI().getRawPath();
            String proxyPrefix = deviceProxyGatewayProperties.getProxyPrefix();
            rawPath = StringUtils.isEmpty(proxyPrefix) ? rawPath : rawPath.replaceFirst(proxyPrefix, "");
            String tarPath = toUri + rawPath;
            log.info("srcPath: {}, tarPath: {} version : {} ,toUri：{}", req.getURI().toString(), tarPath, version, toUri);
            //获取域名后的path
            URI uri = UriComponentsBuilder.fromHttpUrl(tarPath).build().toUri();
            //重新封装request对象
            ServerHttpRequest request = getServerHttpRequest(req, bodyStr, uri);
            Route route = exchange.getAttribute(GATEWAY_ROUTE_ATTR);
            URI routeUri = route.getUri();
            MultiValueMap<String, String> params = encodeParams(req);
            //目标URI
            URI mergedUrl = UriComponentsBuilder.fromUri(uri)
                    .scheme(routeUri.getScheme())
                    .queryParams(params)
                    //参数编码校验
                    .build(true)
                    .toUri();
            exchange.getAttributes().put(GATEWAY_ROUTE_URI, mergedUrl);
            return chain.filter(exchange.mutate().request(request).build());
        };
    }

    /**
    *  QueryParams参数编码
    */
    private MultiValueMap<String, String> encodeParams(ServerHttpRequest req) {
        MultiValueMap<String, String> params = new LinkedMultiValueMap<>();
        req.getQueryParams().forEach((k, v) -> {
            List<String> ps = Lists.newArrayList();
            if (!CollectionUtils.isEmpty(v)) {
                for (String s : v) {
                    if (StringUtils.isEmpty(s)) {
                        continue;
                    }
                    try {
                        ps.add(URLEncoder.encode(s, "UTF-8"));
                    } catch (UnsupportedEncodingException e) {
                        log.error("encode {}", e.getMessage());
                    }
                }
            }
            params.put(k, ps);
        });
        return params;
    }

    /**
    *  重新封装ServerHttpRequest
    */
    private ServerHttpRequest getServerHttpRequest(ServerHttpRequest req, String bodyStr, URI uri) {
        if (StringUtils.isEmpty(bodyStr)) {
            return req.mutate().uri(uri).build();
        }
        DataBuffer bodyDataBuffer = stringBuffer(bodyStr);
        Flux<DataBuffer> bodyFlux = Flux.just(bodyDataBuffer);
        ServerHttpRequest request = new ServerHttpRequestDecorator(req) {
            @Override
            public HttpHeaders getHeaders() {
                HttpHeaders headers = super.getHeaders();
                HttpHeaders httpHeaders = new HttpHeaders();
                headers.forEach((k, v) -> {
                    if (!HEADER_BODY.equals(k) && !HEADER_VERSION.equals(k)) {
                        httpHeaders.addAll(k, v);
                    }
                });
                return httpHeaders;
            }

            @Override
            public Flux<DataBuffer> getBody() {
                return bodyFlux;
            }
        };
        return request;
    }
