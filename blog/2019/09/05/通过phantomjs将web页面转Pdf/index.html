通过phantomjs将web页面转Pdf

Posted by jamie Blog on 2019-09-05 · 阅读 -- 次

1.前言

公司项目上有个新的需求，就是用户点击下载，要将当前的web页面下载为pdf文件，由于这个页面都是各种决策报表，所以导出为pdf，个人感觉还是有必要的，今天写这篇博客，也是更多的是由于以前的工作就算涉及到文件到导出，都是导出excel文件，还没有涉及到pdf，所以这也算是我的一个知识盲区吧！

2.使用步骤

在java中实现将web页面导出为pdf目前也是有多种解决方案。具体有哪些，这边不做陈述，目前这边采用的是phantomjs。

①.下载phantomjs

这个直接从官网下载即可，然后放在项目resources目录下。

@Service
public class PhantomServiceImpl implements PhantomService {

    private static final Logger logger = LoggerFactory.getLogger(PhantomServiceImpl.class);
    private File phantomFile;
    private File rasterizeFile;

    @Value("${isAlpine:}")
    private String isAlpine;

    @PostConstruct
    public void init() throws IOException {
        String plugin = "phantomjs";
        String os = System.getProperty("os.name").toLowerCase();
        if (os.contains("windows")) {
            plugin = "phantomjs.exe";
            phantomFile = File.createTempFile("plutus", "phantom");
            FileUtils.copyInputStreamToFile(ResourceUtils.getURL("classpath:" + "phantomjs/" + plugin).openStream(),
                    phantomFile);
        }else {
            if (isAlpine != null && isAlpine.toLowerCase().equals("true")) {
                if (logger.isWarnEnabled()) {
                    logger.warn("当前系统为alpine,将不创建phantomjs临时文件,使用/usr/share/phantomjs");
                }
                phantomFile = Paths.get("/usr/share/phantomjs").toFile();
            } else {
                phantomFile = File.createTempFile("plutus", "phantom");
                FileUtils.copyInputStreamToFile(ResourceUtils.getURL("classpath:" + "phantomjs/" + plugin).openStream(),
                        phantomFile);
            }
        }
        rasterizeFile = File.createTempFile("plutus", "phantom");
        FileUtils.copyInputStreamToFile(ResourceUtils.getURL("classpath:phantomjs/rasterize.js").openStream(),
                rasterizeFile);
        if (!phantomFile.canExecute()) {
            phantomFile.setExecutable(true);
        }

    }

    @PreDestroy
    public void destroy() {
        phantomFile.delete();
        rasterizeFile.delete();
    }

    public String captureImg(String url) throws IOException {

        ProcessBuilder processBuilder = new ProcessBuilder();
        String img = FileUtils.getTempDirectory().getAbsolutePath() + "/" + System.currentTimeMillis() + ".pdf";

        if (!phantomFile.exists()) {
            logger.error("phantomjs执行文件不存在，pdf无法生成！！！ {}", phantomFile.getAbsolutePath());
        }

        if (!phantomFile.canExecute()) {
            if (logger.isWarnEnabled()) {
                logger.warn("为phantomjs执行文件增加执行权 {}", phantomFile.getAbsolutePath());
            }
            phantomFile.setExecutable(true);
        }
        processBuilder.command(phantomFile.getAbsolutePath(), rasterizeFile.getAbsolutePath(), url, img);
        Process process = processBuilder.start();

        BufferedReader input = new BufferedReader(new InputStreamReader(process.getInputStream()));
        String line;
        StringBuilder msg = new StringBuilder();
        while ((line = input.readLine()) != null) {
            msg.append(line);
            if (logger.isWarnEnabled()) {
                logger.warn(line);
            }
        }
        if (logger.isWarnEnabled()) {
            logger.warn("captureImg:" + img);
        }
        return img;
    }

}
